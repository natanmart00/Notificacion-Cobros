/** actualiza las opciones del formulario de google forms con las opciones 
en tiempo real de nuestra hoja de calculo */

function actualizarFormulario() {
  // lo encuentras en la url del formulario: docs.google.com/forms/d/[este-id]/edit
  const FORM_ID = "179dD058KJuP0DF3Zuq45zs7_A6r9-sdRsGUiuq6KJF0";
  
  // obtiene la hoja de calculo actualmente activa
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  // abre el formulario usando su id
  const form = FormApp.openById(FORM_ID);
  
  // negocio: mantener las opciones del formulario alineadas con usuarios activos
  // framework: los datos en hojas pueden contener encabezados y tipos mixtos
  // codigo: comparaciones debiles o valores no normalizados omitian registros
  // resolucion: filtra por valor booleano o texto y mapea en formato "id - nombre"
  const hojaUsuarios = ss.getSheetByName("usuarios");
  const datosUsuarios = hojaUsuarios.getDataRange().getValues();
  const listaUsuarios = datosUsuarios.slice(1)
    .filter(fila => fila[3] === true || String(fila[3]).trim().toUpperCase() === "TRUE") 
    .map(fila => fila[0] + " - " + fila[1]);

  // negocio: mantener las opciones del formulario alineadas con servicios activos
  // framework: la hoja servicios contiene identificador y estado en cuarta columna
  // codigo: cadenas con espacios o mayusculas inconsistentes fallaban al filtrar
  // resolucion: normaliza el valor de la columna a texto en mayusculas para comparar
  const hojaServicios = ss.getSheetByName("servicios");
  const datosServicios = hojaServicios.getDataRange().getValues();
  const listaServicios = datosServicios.slice(1)
    .filter(fila => fila[3] === true || String(fila[3]).trim().toUpperCase() === "TRUE")
    .map(fila => fila[0] + " - " + fila[1]);

  // obtiene todas las preguntas del formulario
  const items = form.getItems();
  
  items.forEach(item => {
    const titulo = item.getTitle();
    if (titulo === "Usuario") {
      item.asListItem().setChoiceValues(listaUsuarios);
    } else if (titulo === "Servicio") {
      item.asListItem().setChoiceValues(listaServicios);
      // asegura que el campo servicio en el formulario no sea obligatorio
      item.asListItem().setRequired(false);
    }
  });

  SpreadsheetApp.getUi().alert("✅ Formulario actualizado correctamente.");
}

/** funcion que se ejecuta automaticamente al enviar formulario */
function alEnviarFormulario(e) {
  // negocio: registrar pagos recibidos desde el formulario de google forms
  // framework: el disparador automatico inyecta las respuestas en la variable "e"
  // codigo: ejecucion manual directa o estructura invalida generaba error fatal
  // resolucion: valida la existencia del evento antes de leer las respuestas
  if (!e || !e.namedValues) {
    return;
  }

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const hojaMovimientos = ss.getSheetByName("movimientos");
  const respuestas = e.namedValues;

  const usuarioTexto = (respuestas["Usuario"] && respuestas["Usuario"][0]) || "";
  const servicioTexto = (respuestas["Servicio"] && respuestas["Servicio"][0]) || "";
  const monto = (respuestas["Monto"] && respuestas["Monto"][0]) || "";
  const fecha = new Date();
  
  // negocio: el usuario y monto son obligatorios para imputar el pago recibido
  // framework: al desactivar obligatoriedad en formulario servicio puede llegar vacio
  // codigo: condicion previa requeria servicioTexto descartando pagos validos
  // resolucion: solo exige usuario y monto procesando servicio como opcional
  if (!usuarioTexto || !monto) {
    return;
  }

  const usuarioId = parseInt(usuarioTexto.split(" - ")[0]);
  const servicioId = servicioTexto ? parseInt(servicioTexto.split(" - ")[0]) : null;
  
  const hojaSubs = ss.getSheetByName("suscripciones");
  const datosSubs = hojaSubs.getDataRange().getValues();
  let subId = "";
  
  // si indico servicio busca coincidencia exacta con usuario y servicio
  if (servicioId !== null && !isNaN(servicioId)) {
    for (let i = 1; i < datosSubs.length; i++) {
      if (parseInt(datosSubs[i][1]) === usuarioId && parseInt(datosSubs[i][2]) === servicioId) {
        subId = datosSubs[i][0];
        break;
      }
    }
  }

  // si no indico servicio o no hubo coincidencia usa primer suscripcion del usuario
  if (subId === "") {
    for (let i = 1; i < datosSubs.length; i++) {
      if (parseInt(datosSubs[i][1]) === usuarioId) {
        subId = datosSubs[i][0];
        break;
      }
    }
  }

  // inserta fila de pago en la hoja movimientos
  if (subId !== "") {
    hojaMovimientos.appendRow([
      "",
      Utilities.formatDate(fecha, "GMT-6", "dd/MM/yyyy"),
      "",
      subId,
      "PAGO",
      monto,
      "FORMULARIO"
    ]);
  }
}

/** agrega menu en la hoja de calculo */
function onOpen() {
  const ui = SpreadsheetApp.getUi();
  ui.createMenu("🛠️ Automatización")
    .addItem("Sincronizar Formulario", "actualizarFormulario")
    .addToUi();
}
